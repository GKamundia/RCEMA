    in_table = False
    formatted = []
    table_buffer = []

        if line.startswith('|') and '---' in line:

from typing import Dict, List, Tuple
from transformers import AutoTokenizer, TapasTokenizer

class HybridTokenizer:
    """Handles both text and table tokenization"""
    
    def __init__(self,
                 text_model: str = "sentence-transformers/all-mpnet-base-v2",
                 table_model: str = "google/tapas-base"):
        # Text tokenizer
        self.text_tokenizer = AutoTokenizer.from_pretrained(text_model)
        
        # Table tokenizer (TAPAS)
        self.table_tokenizer = TapasTokenizer.from_pretrained(table_model)
        
        # Shared config
        self.max_length = 512  # Max for tabular data

    def tokenize_text(self, text: str) -> List[str]:
        """Tokenize regular text content"""
        return self.text_tokenizer.tokenize(text, truncation=True, max_length=self.max_length)

    def tokenize_table(self, table_html: str) -> Dict:
        """Tokenize tables with TAPAS format"""
        return self.table_tokenizer(
            table=table_html,
            queries=[""],
            max_length=self.max_length,
            return_tensors="pt",
            truncation=True
        )

    def encode_text(self, text: str) -> List[int]:
        """Encode text chunks without special tokens"""
        return self.text_tokenizer.encode(
            text,
            add_special_tokens=False,
            truncation=True,
            max_length=self.max_length
        )

    def encode_table(self, table_html: str) -> Dict:
        """Full table encoding pipeline"""
        return self.tokenize_table(table_html)

    def get_vocab(self) -> Dict:
        """Combine both tokenizers' vocabularies"""
        return {
            **self.text_tokenizer.get_vocab(),
            **self.table_tokenizer.get_vocab()
        }

    @property
    def vocab_size(self) -> int:
        """Total hybrid vocabulary size"""
        return len(self.get_vocab())

from transformers import AutoTokenizer, TapasTokenizer

class HybridTokenizer:
    """Handles both text and table tokenization with dual models"""
    
    def __init__(self,
                 text_model: str = "sentence-transformers/all-mpnet-base-v2",
                 table_model: str = "google/tapas-base"):
        self.text_tokenizer = AutoTokenizer.from_pretrained(text_model)
        self.table_tokenizer = TapasTokenizer.from_pretrained(table_model)
        self.max_text_length = 512
        self.max_table_length = 512

    def tokenize_text(self, text: str) -> list:
        """Tokenize regular text content"""
        return self.text_tokenizer(
            text,
            truncation=True,
            max_length=self.max_text_length,
            return_tensors="pt"
        )

    def tokenize_table(self, table_html: str) -> dict:
        """Tokenize tables with TAPAS processor"""
        return self.table_tokenizer(
            table=table_html,
            queries=[""],
            max_length=self.max_table_length,
            padding="max_length",
            return_tensors="pt"
        )

    def get_combined_vocab(self) -> dict:
        """Merge vocabularies from both tokenizers"""
        return {
            **self.text_tokenizer.get_vocab(),
            **self.table_tokenizer.get_vocab()
        }

    @property
    def text_vocab_size(self) -> int:
        return self.text_tokenizer.vocab_size

    @property
    def table_vocab_size(self) -> int:
        return self.table_tokenizer.vocab_size

def process_pdfs(pdf_paths: List[str]):
    """Main PDF processing pipeline"""
    tokenizer = HybridTokenizer()

        vector: Vector(768) = func.VectorField()

import streamlit as st
import lancedb

from transformers import TapasForQuestionAnswering, TapasTokenizer
from typing import List, Dict, Any
from tokenizer import HybridTokenizer

# Load environment variables
load_dotenv()

class TableQAEngine:
    """Handles table question answering with TAPAS model"""
    def __init__(self):
        self.model = TapasForQuestionAnswering.from_pretrained("google/tapas-base-finetuned-wtq")
        self.tokenizer = TapasTokenizer.from_pretrained("google/tapas-base-finetuned-wtq")
    
    def query_table(self, table_html: str, question: str) -> Dict[str, Any]:
        """Execute QA on HTML tables"""
        try:
            inputs = self.tokenizer(
                table=table_html,
                queries=[question],
                padding="max_length",
                return_tensors="pt",
                truncation=True
            )
            outputs = self.model(**inputs)
            answer = self.tokenizer.convert_logits_to_answer(
                inputs,
                outputs.logits.detach().numpy()
            )
            return {
                "answer": answer[0],
                "confidence": outputs.logits.softmax(dim=1).max().item(),
                "source": "table"
            }
        except Exception as e:
            return {"error": str(e)}

class TextQAEngine:
    """Handles text-based question answering"""
    def __init__(self):
        self.llm = HuggingFaceEndpoint(
            repo_id="mistralai/Mixtral-8x7B-Instruct-v0.1",
            task="text-generation",
            temperature=0.3
        )
    
    def query_text(self, context: str, question: str) -> Dict[str, Any]:
        """Generate answers from text context"""
        prompt = f"""Use only the provided context to answer. If unsure, say so.
        
        Context: {context}
        
        Question: {question}
        Answer:"""
        
        try:
            response = self.llm(
                prompt,
                max_new_tokens=200,
                stop=["\n"]
            )
            return {
                "answer": response[0]['generated_text'],
                "confidence": 1.0,
                "source": "text"
            }
        except Exception as e:
            return {"error": str(e)}

@st.cache_resource
def init_database():
    """Initialize LanceDB connection"""
    db = lancedb.connect("data/lancedb")

def format_table_response(table_html: str, metadata: Dict) -> str:
    """Convert HTML table to styled markdown"""
    df = pd.read_html(io.StringIO(table_html))[0]
    markdown = [
        f"**Source:** {metadata.get('filename', 'Unknown')}",
        f"**Page:** {metadata.get('page', 'N/A')}",
        f"**Table Title:** {metadata.get('title', 'Untitled')}",
        "\n```markdown\n" + df.to_markdown(index=False) + "\n```"
    ]
    return "\n".join(markdown)

def format_text_response(text: str, metadata: Dict) -> str:
    """Format text chunks with metadata"""
    return "\n".join([
        f"**Source:** {metadata.get('filename', 'Unknown')}",
        f"**Pages:** {', '.join(map(str, metadata.get('page_numbers', [])))}",
        f"**Section:** {metadata.get('headings', [''])[0]}",
        "---",
        text
    ])

def get_context(table, query: str, max_results: int = 3) -> List[Dict]:
    """Retrieve relevant context from database"""
    return table.search(query) \
               .limit(max_results) \
               .to_pandas() \
               .to_dict('records')
